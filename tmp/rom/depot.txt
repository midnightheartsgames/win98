  function romanDepotPose(dt, v) {
    const wake = ease(clamp((dt - 0.85) / 0.55, 0, 1));
    const talk = clamp((dt - 1.6) / 0.5, 0, 1);
    const sway = talk > 0 ? S(dt * 3.1) * 0.05 : 0;
    const gesture = talk * (0.5 + 0.5 * S(dt * 5.2));
    const mouth = dt > 1.72 && v > 0.22 ? (v > 0.55 ? 'wide' : 'open') : 'shut';
    return Object.assign({
      pos: [0.1, 0.5, -0.12],
      yaw: lerp(-0.1, -0.34, wake) + sway,
      lean: 0,
      roll: sway * 0.5,
      sp: [lerp(0.62, 0.06, wake), lerp(0, -0.1, wake), sway],
      nk: [lerp(0.72, 0.02, wake), lerp(0, -0.16, wake), sway * 0.6],
      hL: [0.3, lerp(0.02, 0.1 + gesture * 0.22, wake), lerp(0.38, 0.42 + gesture * 0.1, wake)],
      hR: [-0.3, lerp(0.02, 0.06 + gesture * 0.18, wake), lerp(0.38, 0.4 + gesture * 0.08, wake)],
      eL: [0.45, -0.1, 0.0],
      eR: [-0.45, -0.1, 0.0],
      gL: wake > 0.5 ? 'open' : 'relax',
      gR: wake > 0.5 ? 'open' : 'relax',
      eyes: dt < 0.9 ? 'shut' : dt < 1.5 ? 'wide' : 'open',
      mouth
    }, SIT_LEGS);
  }

  function nikoDepotPose(dt, v) {
    const reach = ease(clamp((dt - 0.9) / 0.5, 0, 1)) * (1 - ease(clamp((dt - 2.3) / 0.5, 0, 1)));
    const mouth = dt < 1.72 && v > 0.2 ? (v > 0.52 ? 'wide' : 'open') : 'shut';
    return Object.assign({
      pos: [-0.3, 0.84, 0.18],
      yaw: 1.3,
      lean: 0,
      roll: 0,
      sp: [0.16, -0.1, 0],
      nk: [0.08, 0.1, 0],
      hL: [0.26, -0.18, 0.12],
      hR: [-0.28, -0.04 + reach * 0.3, 0.2 + reach * 0.34],
      eL: [0.42, -0.1, -0.08],
      eR: [-0.42, -0.1, 0.1],
      gL: 'relax',
      gR: reach > 0.3 ? 'open' : 'relax',
      eyes: 'open',
      mouth
    }, NIKO_LEGS);
  }

  function stageDepot(t, shot, v) {
    if (shot === 6) {
      const dt = t - 10.95;
      return {
        t,
        shot,
        set: 'depot',
        env: ENV_DEPOT,
        blur: 0,
        fade: 1,
        eyelids: [0, 0],
        subtitle: 'sub1',
        camera: {
          eye: [-0.02 + dt * 0.006, 0.92, 0.8 - dt * 0.012],
          at: [0.02 + dt * 0.004, 0.9, 0],
          up: [0, 1, 0],
          fov: 42
        },
        cast: [
          { who: 'roman', pose: romanDepotPose(dt, v) },
          { who: 'niko', pose: nikoDepotPose(dt, v) }
        ]
      };
    }

    const dt = t - 14.25;
    const nikoPose = Object.assign({
      pos: [-1.0, 0.6, 0.18],
      yaw: 1.0,
      lean: 0,
      roll: 0,
      sp: [0.04, 0, 0],
      nk: [0.1, 0.1 + dt * 0.2, 0],
      hL: [0.22, -0.18, 0.08],
      hR: [-0.22, -0.18, 0.08],
      eL: [0.42, -0.1, -0.08],
      eR: [-0.42, -0.1, -0.08],
      gL: 'relax',
      gR: 'relax',
      eyes: 'open',
      mouth: v > 0.25 ? 'open' : 'shut'
    }, NIKO_LEGS);

    return {
      t,
      shot,
      set: 'depot',
      env: ENV_DEPOT,
      blur: 0,
      fade: 1,
      eyelids: [0, 0],
      subtitle: 'sub1',
      camera: {
        eye: [-0.4, 1.1, 0.18],
        at: [-1.0, 1.1, -0.04],
        up: [0, 1, 0],
        fov: 36
      },
      cast: [
        { who: 'niko', pose: nikoPose }
      ]
    };
  }
